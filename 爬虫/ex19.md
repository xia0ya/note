# 练习十九：JS加密混淆与逆向调试基础

本练习演示如何处理网页中的JS加密混淆数据，利用Selenium获取渲染后源码，并逆向分析JS变量。

---

```python
# coding=utf-8
# 功能：处理JS加密混淆，逆向调试并提取数据

import requests
from lxml import etree
from selenium import webdriver
import time
from selenium.webdriver.chrome.service import Service

base_url = 'https://spiderbuf.cn/web-scraping-practice/javascript-confuse-encrypt-reverse'

myheaders = {
    'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/91.0.4472.164 Safari/537.36'}

def getHTML(url, file_name=''):
    driver_path = r"D:\Program Files\chromedriver\chromedriver-win64\chromedriver.exe"  # 替换为实际路径
    service = Service(driver_path)
    client = webdriver.Chrome(service=service)
    client.get(url)
    time.sleep(3)
    html = client.page_source   # 获取浏览器渲染后的源码
    print(html)
    client.quit()
    if file_name != '':
        with open(file_name, 'w', encoding='utf-8') as f:
            f.write(html)
    return html

def parseHTML(html, file_name=''):
    root = etree.HTML(html)
    trs = root.xpath('//tr')
    if file_name != '':
        f = open(file_name, 'w', encoding='utf-8')
    for tr in trs:
        tds = tr.xpath('./td')
        s = ''
        for td in tds:
            s = s + str(td.xpath('string(.)')) + '|'
        print(s)
        if (s != '') & (file_name != ''):
            f.write(s + '\n')
    if file_name != '':
        f.close()

if __name__ == '__main__':
    # 示例1：用Selenium获取渲染后源码并解析
    # html = getHTML(base_url, './h04.html')
    # parseHTML(html, './h04.txt')

    # 示例2：逆向分析JS变量
    url = 'https://spiderbuf.cn/static/js/h04/udSL29.js'
    js_code = requests.get(url, headers=myheaders).text
    a = js_code.index('=') + 1
    b = js_code.index(';')
    js_code = js_code[a:b]
    # 将字符串转换为字典
    dict_data = eval(js_code)
    print(dict_data)
    for item in dict_data:
        print(item)
```

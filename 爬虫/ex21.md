# 练习二十一：初识浏览器指纹与Selenium反爬机制

本练习演示如何通过Selenium配置规避部分浏览器指纹检测，分析Selenium被反爬的原理，并抓取网页数据。

---

```python
# coding=utf-8
# 功能：分析Selenium被反爬的原理，规避部分浏览器指纹检测

import base64
import hashlib
import time
import requests
from lxml import etree
from selenium import webdriver
from selenium.webdriver.chrome.service import Service

base_url = 'https://spiderbuf.cn/web-scraping-practice/selenium-fingerprint-anti-scraper'

myheaders = {
    'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/91.0.4472.164 Safari/537.36'}

def getHTML(url, file_name=''):
    driver_path = r"D:\Program Files\chromedriver\chromedriver-win64\chromedriver.exe"  # 替换为实际路径
    service = Service(driver_path)
    options = webdriver.ChromeOptions()
    options.add_argument('disable-infobars')
    # options.add_argument('headless')  # 可选：无头模式
    options.set_capability('goog:loggingPrefs', {'browser': 'ALL'})  # 输出浏览器console日志
    options.add_argument('--disable-blink-features=AutomationControlled')  # 改变navigator.webdriver属性值
    client = webdriver.Chrome(service=service, options=options)
    client.get(url)
    time.sleep(5)
    print(client.page_source)
    html = client.page_source
    # client.quit()
    if file_name != '':
        with open(file_name, 'w', encoding='utf-8') as f:
            f.write(html)
    return html

def parseHTML(html, file_name=''):
    root = etree.HTML(html)
    trs = root.xpath('//tr')
    if file_name != '':
        f = open(file_name, 'w', encoding='utf-8')
    for tr in trs:
        tds = tr.xpath('./td')
        s = ''
        for td in tds:
            s = s + str(td.xpath('string(.)')) + '|'
        print(s)
        if (s != '') & (file_name != ''):
            f.write(s + '\n')
    if file_name != '':
        f.close()

if __name__ == '__main__':
    # 示例1：用Selenium规避指纹检测并抓取数据
    html = getHTML(base_url, './h06.html')
    print(html)
    parseHTML(html, './h06.txt')
```

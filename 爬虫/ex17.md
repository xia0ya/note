# 练习十七：网页滚动加载与逆向基础

本练习演示如何处理网页滚动加载（如"下拉加载更多"），并逆向分析分页参数，批量提取电影信息。

```python
# coding=utf-8
# 功能：处理网页滚动加载与逆向分页，批量提取电影信息

import os.path
import requests
from lxml import etree
import time

base_url = 'https://spiderbuf.cn/web-scraping-practice/scraping-scroll-load'

myheaders = {
    'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/91.0.4472.164 Safari/537.36'}

def getHTML(url, file_name=''):
    html = requests.get(url, headers=myheaders).text
    if file_name != '':
        with open(file_name, 'w', encoding='utf-8') as f:
            f.write(html)
    return html

def downloadImage(url, path=''):
    img_data = requests.get(url, headers=myheaders).content
    file_name = url.split('/').pop()
    with open(os.path.join(path, file_name), 'wb') as img:
        img.write(img_data)

def parseHTML(html):
    root = etree.HTML(html)
    divs = root.xpath('/html/body/div/div/div[@style="margin-top: 10px;"]')
    i = 1
    for div in divs:
        if i % 2 == 0:
            summarys = div.xpath('./div/text()')
            summary = summarys[0].strip() if summarys else ''
            print('简介:', summary)
        else:
            titles = div.xpath('./div/h2')
            title = titles[0].text if titles else ''
            print('标题:', title)
            img_urls = div.xpath('./div/div/img/@src')
            img_url = 'https://spiderbuf.cn/' + img_urls[0] if img_urls else ''
            print('图片链接:', img_url)
            # downloadImage(img_url, './data/h03')  # 可选：下载图片
            ratings = div.xpath('./div/div/span[contains(text(),"豆瓣电影评分:")]/following::text()[1]')
            rating = ratings[0].strip() if ratings else ''
            print('评分:', rating)
            directors = div.xpath('./div/div/span/span[contains(text(),"导演")]/following::text()')
            director = directors[1].strip() if len(directors) > 1 else ''
            if len(directors) > 3:
                director += '/' + directors[2].strip()
            print('导演:', director)
            scriptwriters = div.xpath('./div/div/span/span[contains(text(),"编剧")]/following::text()')
            scriptwriter = scriptwriters[1].strip() if len(scriptwriters) > 0 else ''
            if len(scriptwriters) > 3:
                scriptwriter += scriptwriters[2].strip()
            print('编剧:', scriptwriter)
            performers = div.xpath('./div/div/span/span[contains(text(),"主演")]/following::text()')
            performer = performers[1].strip() if len(performers) > 0 else ''
            if len(performers) > 3:
                performer += performers[2].strip()
            print('主演:', performer)
            genres = div.xpath('./div/div/span/span[contains(text(),"类型:")]/following::text()')
            genre = genres[0].strip() if genres else ''
            if len(performers) > 1:
                genre += genres[1].strip()
            print('类型:', genre)
            areas = div.xpath('./div/div/span/span[contains(text(),"制片国家/地区:")]/following::text()')
            area = areas[0].strip() if areas else ''
            print('制片国家/地区:', area)
            langs = div.xpath('./div/div/span/span[contains(text(),"语言:")]/following::text()')
            lang = langs[0].strip().replace('\n', '') if langs else ''
            if len(langs) > 1:
                lang += langs[1].strip().replace('\n', '')
            print('语言:', lang)
            aliases = div.xpath('./div/div/span/span[contains(text(),"又名:")]/following::text()')
            alias = aliases[0].strip().replace('\n', '').replace('|', '') if aliases else ''
            if len(aliases) > 1:
                alias += aliases[1].strip().replace('\n', '').replace('|', '')
            print('又名:', alias)
            imdbs = div.xpath('./div/div/span[contains(text(),"IMDb:")]/following::text()')
            imdb = imdbs[0].strip().replace('\n', '') if imdbs else ''
            print('IMDb:', imdb)
            release_dates = div.xpath('./div/div/span/span[contains(text(),"上映日期:")]/following::text()')
            release_date = release_dates[0].strip().replace('\n', '') if release_dates else ''
            if len(release_dates) > 1:
                release_date += release_dates[1].strip().replace('\n', '')
            print('上映日期:', release_date)
            runtimes = div.xpath('./div/div/span/span[contains(text(),"片长:")]/following::text()')
            runtime = runtimes[0].strip().replace('\n', '') if runtimes else ''
            if len(runtimes) > 1:
                runtime += runtimes[1].strip().replace('\n', '')
            print('片长:', runtime)
        i += 1

if __name__ == '__main__':
    html = getHTML(base_url, './data/h03/h03.html')
    # 获取下一页uri
    uri = ''
    root = etree.HTML(html)
    divs = root.xpath('//div[@id="sLaOuol2SM0iFj4d"]/text()')
    if len(divs) > 0:
        uri = divs[0]
    i = 1
    while (uri != '') & (i < 10):
        print(uri)
        html = getHTML(base_url + '/' +  uri, f'./data/h03/h03_{uri}.html')
        uri = ''
        root = etree.HTML(html)
        divs = root.xpath('//div[@id="sLaOuol2SM0iFj4d"]/text()')
        if len(divs) > 0:
            uri = divs[0]
        i += 1
```

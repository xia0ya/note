# 练习十：用户名密码+验证码登录后爬取数据

本练习演示如何通过用户名、密码和验证码（或Cookie）登录网站后，抓取网页表格数据。验证码处理可进一步扩展为手动输入或自动识别。

---

```python
# encoding: utf-8
# @Author: xia0ya
# 功能：通过用户名、密码和验证码（或Cookie）登录后抓取网页表格数据

import requests
from lxml import etree

# 目标URL
url = "https://spiderbuf.cn/web-scraping-practice/web-scraping-with-captcha/list"
payload = {'username':'admin','password':'123456'}
# 自定义请求头（包含Cookie信息）
MyHeaders = {
    'user-agent': "Mozilla/5.0 (Linux; Android 6.0; Nexus 5 Build/MRA58N) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/137.0.0.0 Mobile Safari/537.36 Edg/137.0.0.0",
    'Cookie':'_ga=GA1.1.786419057.1750067586; __gads=ID=66e95ea031935bfa:T=1750067589:RT=1750085249:S=ALNI_MZfRLh8tUkKsTb8E6URgp6_ST5_wg; __gpi=UID=00001130564aff80:T=1750067589:RT=1750085249:S=ALNI_Ma--kgHOowXCWNqqKm8h2luvnHRSA; __eoi=ID=72c8e5dc4a35b630:T=1750067589:RT=1750085249:S=AA-AfjYRPzKyu5es8xaQM3hyozKp; _ga_7B42BKG1QE=GS2.1.s1750081663$o3$g1$t1750085527$j60$l0$h0; admin=2bfff62d143f054f42f47c7c6116f768'
}

# 发送GET请求（带上Cookie和参数）
response = requests.get(url, headers=MyHeaders, data=payload)

# 检查响应状态
if response.status_code == 200:
    html = response.text
    # print(html)  # 可选：打印获取到的网页源码

    # 解析HTML内容
    root = etree.HTML(html)
    trs = root.xpath('//tr')

    # 遍历表格行并提取数据
    for tr in trs:
        tds = tr.xpath('./td/text()')  # 获取每个单元格的文本
        s = '|'.join(tds)  # 将单元格内容用'|'拼接
        print(s)  # 打印每行数据

    # 保存文件（可选）
    # with open("data01.txt", "w", encoding='utf-8') as f:
    #     for tr in trs:
    #         tds = tr.xpath('./td/text()')
    #         s = '|'.join(tds)
    #         if s:
    #             f.write(s + '\n')
else:
    print(f"请求失败，状态码：{response.status_code}")
```
